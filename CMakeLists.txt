cmake_minimum_required(VERSION 3.22.1)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(redsocks C)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(TLS_BACKEND openssl CACHE STRING "libevent TLS backend: openssl")
set_property(CACHE TLS_BACKEND PROPERTY STRINGS openssl)
if(DEFINED PROXYDROID_TLS_BACKEND)
    message(FATAL_ERROR "PROXYDROID_TLS_BACKEND was removed; use TLS_BACKEND=openssl")
endif()
if(NOT TLS_BACKEND STREQUAL "openssl")
    message(FATAL_ERROR "This redsocks build requires libevent's OpenSSL backend (TLS_BACKEND=openssl)")
endif()
if(EVENT__DISABLE_OPENSSL)
    message(FATAL_ERROR "TLS_BACKEND=openssl conflicts with EVENT__DISABLE_OPENSSL=ON")
endif()
if(TARGET event_core AND TARGET event_openssl)
    set(REDSOCKS_EVENT_LIBRARY event_openssl event_core)
elseif(TARGET event_core OR TARGET event_openssl OR TARGET ProxyDroid::TLS)
    message(FATAL_ERROR "The parent must supply both event_core and event_openssl for the selected TLS backend")
else()
    # Standalone component builds deliberately use the system dependency pair.
    find_path(REDSOCKS_EVENT_INCLUDE_DIR event2/event.h REQUIRED)
    find_library(REDSOCKS_EVENT_CORE NAMES event_core REQUIRED)
    find_library(REDSOCKS_EVENT_SSL NAMES event_openssl REQUIRED)
    include(CheckCSourceCompiles)
    include(CMakePushCheckState)
    cmake_push_check_state(RESET)
    set(CMAKE_REQUIRED_INCLUDES "${REDSOCKS_EVENT_INCLUDE_DIR}")
    check_c_source_compiles("#include <event2/event.h>\n#if LIBEVENT_VERSION_NUMBER < 0x02010100\n#error libevent 2.1.1 or later required\n#endif\nint main(void) { return 0; }" REDSOCKS_HAVE_LIBEVENT_2_1)
    cmake_pop_check_state()
    if(NOT REDSOCKS_HAVE_LIBEVENT_2_1)
        message(FATAL_ERROR "redsocks requires libevent >= 2.1.1 with OpenSSL support")
    endif()
    message(STATUS "Standalone redsocks: system libevent/OpenSSL")
    set(REDSOCKS_EVENT_LIBRARY ${REDSOCKS_EVENT_SSL} ${REDSOCKS_EVENT_CORE})
endif()
if(NOT TARGET ProxyDroid::TLS)
    find_package(OpenSSL 1.1.1 REQUIRED)
    add_library(redsocks-system-tls INTERFACE)
    target_link_libraries(redsocks-system-tls INTERFACE OpenSSL::SSL OpenSSL::Crypto)
    add_library(ProxyDroid::TLS ALIAS redsocks-system-tls)
    add_library(ProxyDroid::Crypto ALIAS OpenSSL::Crypto)
endif()
find_package(Threads REQUIRED)
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(USE_IPTABLES 1)
endif()
configure_file(cmake/config.h.in generated/config.h)
set(CNTLM_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../cntlm" CACHE PATH "Embedded cntlm source directory")
if(NOT EXISTS "${CNTLM_SOURCE_DIR}/embedded.h")
    message(FATAL_ERROR "cntlm is missing; initialize ProxyDroid's submodules or set CNTLM_SOURCE_DIR")
endif()
if(NOT TARGET cntlm-embedded)
    add_subdirectory("${CNTLM_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/cntlm")
endif()
file(STRINGS "${CNTLM_SOURCE_DIR}/VERSION" REDSOCKS_CNTLM_VERSION LIMIT_COUNT 1)
set(REDSOCKS_VERSION_FILE "${CMAKE_CURRENT_BINARY_DIR}/generated/version.c")
add_custom_target(redsocks-version
    COMMAND "${CMAKE_COMMAND}" "-DREDSOCKS_SOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}"
        "-DREDSOCKS_VERSION_OUTPUT=${REDSOCKS_VERSION_FILE}"
        "-DREDSOCKS_SOURCE_INFO_SCRIPT=${REDSOCKS_SOURCE_INFO_SCRIPT}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/version.cmake"
    BYPRODUCTS "${REDSOCKS_VERSION_FILE}" VERBATIM)
add_library(cntlm-adapter OBJECT cntlm-embed.c)
set_target_properties(cntlm-adapter PROPERTIES C_STANDARD 99 C_VISIBILITY_PRESET hidden)
foreach(property COMPILE_DEFINITIONS COMPILE_OPTIONS INCLUDE_DIRECTORIES)
    get_target_property(cntlm_${property} cntlm-embedded ${property})
    set_property(TARGET cntlm-adapter PROPERTY ${property} "${cntlm_${property}}")
endforeach()
target_include_directories(cntlm-adapter PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" ${REDSOCKS_EVENT_INCLUDE_DIR})
target_link_libraries(cntlm-adapter PRIVATE ProxyDroid::TLS ${REDSOCKS_EVENT_LIBRARY})
set(redsocks_udp_sources dnstc.c)
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # IP_TRANSPARENT and TCP Fast Open are Linux-specific in these modules.
    set(redsocks_udp_sources redudp.c dnstc.c dnsu2t.c)
endif()
add_library(redsocks-core SHARED engine.c cntlm-io.c upstream-tls.c
    parser.c redsocks.c log.c http-connect.c tls-sni.c socks4.c socks5.c http-relay.c
    base.c base64.c md5.c http-auth.c utils.c ${redsocks_udp_sources}
    "${REDSOCKS_VERSION_FILE}" $<TARGET_OBJECTS:cntlm-embedded> $<TARGET_OBJECTS:cntlm-adapter>)
set_target_properties(redsocks-core PROPERTIES OUTPUT_NAME redsocks C_STANDARD 99)
add_dependencies(redsocks-core redsocks-version)
target_compile_definitions(redsocks-core PRIVATE _GNU_SOURCE _DEFAULT_SOURCE _XOPEN_SOURCE=600
    REDSOCKS_CNTLM_VERSION="${REDSOCKS_CNTLM_VERSION}")
if(APPLE)
    target_compile_definitions(redsocks-core PRIVATE _DARWIN_C_SOURCE)
endif()
target_compile_options(redsocks-core PRIVATE -Wall -include "${CMAKE_CURRENT_SOURCE_DIR}/engine-compat.h")
target_include_directories(redsocks-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}"
    PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" ${REDSOCKS_EVENT_INCLUDE_DIR})
target_link_libraries(redsocks-core PRIVATE ${REDSOCKS_EVENT_LIBRARY} ProxyDroid::TLS ProxyDroid::Crypto Threads::Threads m)
if(ANDROID)
    target_sources(redsocks-core PRIVATE jni.c)
    target_link_libraries(redsocks-core PRIVATE log)
    set_property(TARGET redsocks-core APPEND_STRING PROPERTY LINK_FLAGS " -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL")
else()
    add_executable(redsocks main.c)
    target_compile_definitions(redsocks PRIVATE _GNU_SOURCE)
    target_link_libraries(redsocks PRIVATE redsocks-core Threads::Threads)
    include(CTest)
    if(BUILD_TESTING AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_executable(redsocks-test-pulse tests/timing-pulse.c)
        set_target_properties(redsocks-test-pulse PROPERTIES C_STANDARD 99)
        target_compile_options(redsocks-test-pulse PRIVATE -Wall -Wextra)
        add_executable(redsocks-test-deferred-free tests/test_deferred_free.c)
        set_target_properties(redsocks-test-deferred-free PROPERTIES C_STANDARD 99)
        target_compile_options(redsocks-test-deferred-free PRIVATE -Wall -Wextra)
        target_include_directories(redsocks-test-deferred-free PRIVATE
            "${CMAKE_CURRENT_SOURCE_DIR}" ${REDSOCKS_EVENT_INCLUDE_DIR})
        target_link_libraries(redsocks-test-deferred-free PRIVATE
            ${REDSOCKS_EVENT_LIBRARY} Threads::Threads)
        add_test(NAME redsocks-deferred-free COMMAND redsocks-test-deferred-free)
        set_tests_properties(redsocks-deferred-free PROPERTIES TIMEOUT 10)
        find_program(REDSOCKS_TEST_PYTHON NAMES python3 REQUIRED)
        add_executable(redsocks-test-tls-half-close tests/test_tls_half_close.c)
        set_target_properties(redsocks-test-tls-half-close PROPERTIES C_STANDARD 99)
        target_compile_options(redsocks-test-tls-half-close PRIVATE -Wall -Wextra)
        target_include_directories(redsocks-test-tls-half-close PRIVATE
            "${CMAKE_CURRENT_SOURCE_DIR}" ${REDSOCKS_EVENT_INCLUDE_DIR})
        target_link_libraries(redsocks-test-tls-half-close PRIVATE redsocks-core
            ${REDSOCKS_EVENT_LIBRARY} ProxyDroid::TLS ProxyDroid::Crypto Threads::Threads)
        add_test(NAME redsocks-tls-half-close COMMAND "${REDSOCKS_TEST_PYTHON}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_tls_half_close.py"
            "$<TARGET_FILE:redsocks-test-tls-half-close>")
        set_tests_properties(redsocks-tls-half-close PROPERTIES TIMEOUT 10)
    endif()
endif()
