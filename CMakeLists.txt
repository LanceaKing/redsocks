cmake_minimum_required(VERSION 3.22.1)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
    project(redsocks C)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(TARGET event)
    set(REDSOCKS_EVENT_LIBRARY event)
else()
    find_path(REDSOCKS_EVENT_INCLUDE_DIR event.h)
    find_library(REDSOCKS_EVENT_CORE NAMES event_core event)
    find_library(REDSOCKS_EVENT_SSL NAMES event_openssl)
    if(NOT REDSOCKS_EVENT_CORE OR NOT REDSOCKS_EVENT_SSL)
        message(FATAL_ERROR "libevent core and OpenSSL libraries required")
    endif()
    set(REDSOCKS_EVENT_LIBRARY ${REDSOCKS_EVENT_SSL} ${REDSOCKS_EVENT_CORE})
endif()
if(NOT TARGET OpenSSL::SSL)
    find_package(OpenSSL 1.1.1 REQUIRED)
endif()
find_package(Threads REQUIRED)
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(USE_IPTABLES 1)
endif()
configure_file(cmake/config.h.in generated/config.h)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cntlm/embedded.h")
    message(FATAL_ERROR "cntlm submodule missing: run git submodule update --init --recursive")
endif()
add_subdirectory(cntlm)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/cntlm/VERSION" REDSOCKS_CNTLM_VERSION LIMIT_COUNT 1)
set(REDSOCKS_VERSION_FILE "${CMAKE_CURRENT_BINARY_DIR}/generated/version.c")
add_custom_target(redsocks-version
    COMMAND "${CMAKE_COMMAND}" "-DREDSOCKS_SOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}"
        "-DREDSOCKS_VERSION_OUTPUT=${REDSOCKS_VERSION_FILE}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/version.cmake"
    BYPRODUCTS "${REDSOCKS_VERSION_FILE}" VERBATIM)
target_sources(cntlm-embedded PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/cntlm-embed.c")
target_include_directories(cntlm-embedded PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" ${REDSOCKS_EVENT_INCLUDE_DIR})
target_link_libraries(cntlm-embedded PRIVATE OpenSSL::SSL ${REDSOCKS_EVENT_LIBRARY})
add_library(redsocks-core SHARED engine.c cntlm-io.c upstream-tls.c
    parser.c redsocks.c log.c http-connect.c tls-sni.c socks4.c socks5.c http-relay.c
    base.c base64.c md5.c http-auth.c utils.c redudp.c dnstc.c dnsu2t.c
    "${REDSOCKS_VERSION_FILE}" $<TARGET_OBJECTS:cntlm-embedded>)
set_target_properties(redsocks-core PROPERTIES OUTPUT_NAME redsocks C_STANDARD 99)
add_dependencies(redsocks-core redsocks-version)
target_compile_definitions(redsocks-core PRIVATE _GNU_SOURCE _DEFAULT_SOURCE _XOPEN_SOURCE=600
    REDSOCKS_CNTLM_VERSION="${REDSOCKS_CNTLM_VERSION}")
target_compile_options(redsocks-core PRIVATE -Wall -include "${CMAKE_CURRENT_SOURCE_DIR}/engine-compat.h")
target_include_directories(redsocks-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}"
    PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" ${REDSOCKS_EVENT_INCLUDE_DIR})
target_link_libraries(redsocks-core PRIVATE ${REDSOCKS_EVENT_LIBRARY} OpenSSL::SSL OpenSSL::Crypto Threads::Threads m)
if(ANDROID)
    target_sources(redsocks-core PRIVATE jni.c)
    target_link_libraries(redsocks-core PRIVATE log)
    set_property(TARGET redsocks-core APPEND_STRING PROPERTY LINK_FLAGS " -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL")
else()
    add_executable(redsocks main.c)
    target_compile_definitions(redsocks PRIVATE _GNU_SOURCE)
    target_link_libraries(redsocks PRIVATE redsocks-core Threads::Threads)
    include(CTest)
endif()
